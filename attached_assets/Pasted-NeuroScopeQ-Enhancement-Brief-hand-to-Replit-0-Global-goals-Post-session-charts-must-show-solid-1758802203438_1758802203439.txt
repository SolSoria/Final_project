// src/api/report.ts (concept)
import puppeteer from "puppeteer";
export async function renderReport(html: string) {
  const browser = await puppeteer.launch({ args:['--no-sandbox'] });
  const page = await browser.newPage();
  await page.setContent(html, { waitUntil: "networkidle0" });
  const pdf = await page.pdf({ format: "A4", printBackground: true, margin:{ top:"15mm", right:"12mm", bottom:"15mm", left:"12mm" }});
  await browser.close();
  return pdf;
}

Patient list endpoint (server)
router.get("/api/patients", async (req,res)=>{
  await connectDB();
  const { search="", cohort, setting } = req.query as any;
  const q:any = {};
  if (search) q.name = { $regex: new RegExp(search,'i') };
  if (cohort) q.cohort = cohort;
  if (setting) q.setting = setting;
  const pts = await Patient.find(q).sort({ updatedAt:-1 }).limit(200).lean();
  res.json(pts);
});
Download report route
router.get("/api/report/:patientId", async (req,res)=>{
  const p = await Patient.findById(req.params.patientId).lean();
  const s = await Session.find({ patientId: p._id }).sort({ date:1 }).lean();
  const html = renderReportHtml({ patient:p, sessions:s }); // your template fn
  const pdf = await renderReport(html);
  res.setHeader("Content-Type","application/pdf");
  res.setHeader("Content-Disposition", `attachment; filename="report_${p.name}.pdf"`);
  res.send(Buffer.from(pdf));
});
